Assignment 1: Employee Repository
interface EmployeeRepository {

    fun getEmployeeName(
        employeeId: Int
    ): String?

}

class EmployeeService(

    private val repository: EmployeeRepository

) {

    fun getEmployeeDetails(
        employeeId: Int
    ): String {

        val name = repository.getEmployeeName(employeeId)

        return name ?: "Employee Not Found"

    }

}
